# BazaarPlusPlus Installer Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Tauri 2 + Svelte desktop installer for BazaarPlusPlus that detects the environment, installs BepInEx from a bundled zip, and patches Steam's VDF launch options on macOS and Windows.

**Architecture:** Rust backend exposes Tauri commands for detection, BepInEx extraction, and VDF mutation. Core logic lives in pure functions (testable) wrapped by thin Tauri command handlers. Svelte frontend calls `invoke()` and listens to `installer://log` events for real-time progress.

**Tech Stack:** Tauri 2, Svelte + TypeScript, Rust (`keyvalues-parser`, `zip`, `reqwest`, `winreg` on Windows)

**Spec:** `docs/superpowers/specs/2026-03-10-installer-design.md`

```rust
use serde::{Deserialize, Serialize};
use std::path::PathBuf;

#[derive(Debug, Serialize, Deserialize)]
pub struct EnvironmentInfo {
    pub steam_path: Option<String>,
    pub game_path: Option<String>,
    pub dotnet_version: Option<String>,
    pub dotnet_ok: bool,
    pub bepinex_installed: bool,
}

/// Parse libraryfolders.vdf and return the library path that contains app_id.
pub fn find_game_in_library_vdf(vdf_content: &str, app_id: &str) -> Option<String> {
    // Parse with keyvalues_parser; navigate folder entries manually
    use keyvalues_parser::KeyValues;
    let kv = KeyValues::parse(vdf_content).ok()?;
    let folders = kv.value.as_obj()?;
    for (_key, val) in folders {
        let folder = val.as_obj()?;
        // Collect path and apps as plain String lookups to avoid ? in closures
        let mut lib_path: Option<String> = None;
        let mut has_app = false;
        for (k, v) in folder {
            if k.as_str() == "path" {
                lib_path = v.as_str().map(|s| s.to_string());
            }
            if k.as_str() == "apps" {
                if let Some(apps) = v.as_obj() {
                    has_app = apps.iter().any(|(ak, _)| ak.as_str() == app_id);
                }
            }
        }
        if has_app {
            return lib_path;
        }
    }
    None
}

fn get_steam_path() -> Option<PathBuf> {
    #[cfg(target_os = "macos")]
    {
        let p = dirs::home_dir()?.join("Library/Application Support/Steam");
        if p.exists() { return Some(p); }
    }
    #[cfg(target_os = "windows")]
    {
        use winreg::enums::*;
        use winreg::RegKey;
        let hkcu = RegKey::predef(HKEY_CURRENT_USER);
        if let Ok(key) = hkcu.open_subkey(r"Software\Valve\Steam") {
            if let Ok(path) = key.get_value::<String, _>("SteamPath") {
                let p = PathBuf::from(path);
                if p.exists() { return Some(p); }
            }
        }
    }
    None
}

fn get_game_path(steam_path: &PathBuf) -> Option<PathBuf> {
    let lf_path = steam_path.join("steamapps/libraryfolders.vdf");
    let content = std::fs::read_to_string(lf_path).ok()?;
    let library_path = find_game_in_library_vdf(&content, "2138550")?;
    let game_path = PathBuf::from(library_path)
        .join("steamapps/common/The Bazaar");
    if game_path.exists() { Some(game_path) } else { None }
}

#[tauri::command]
pub fn detect_environment() -> Result<EnvironmentInfo, String> {
    let steam_path = get_steam_path();
    let game_path = steam_path.as_ref().and_then(|sp| get_game_path(sp));
    let bepinex_installed = game_path.as_ref()
        .map(|gp| gp.join("BepInEx/core/BepInEx.Core.dll").exists())
        .unwrap_or(false);
    Ok(EnvironmentInfo {
        steam_path: steam_path.map(|p| p.to_string_lossy().into_owned()),
        game_path: game_path.map(|p| p.to_string_lossy().into_owned()),
        dotnet_version: None,
        dotnet_ok: false,
        bepinex_installed,
    })
}
```

```rust
/// Returns highest qualifying version string (NETCore.App >= 6), or None.
pub fn parse_dotnet_runtimes(output: &str) -> Option<String> {
    output.lines()
        .filter(|l| l.starts_with("Microsoft.NETCore.App "))
        .filter_map(|l| l.split_whitespace().nth(1).map(|v| v.to_string()))
        .filter(|v| {
            v.split('.').next()
                .and_then(|major| major.parse::<u32>().ok())
                .map(|major| major >= 6)
                .unwrap_or(false)
        })
        .max()
}

fn detect_dotnet() -> (Option<String>, bool) {
    // On Windows, also try %PROGRAMFILES%\dotnet\dotnet.exe as fallback
    #[cfg(target_os = "windows")]
    let candidates = {
        let mut v = vec!["dotnet".to_string()];
        if let Ok(pf) = std::env::var("PROGRAMFILES") {
            v.push(format!(r"{}\dotnet\dotnet.exe", pf));
        }
        v
    };
    #[cfg(not(target_os = "windows"))]
    let candidates = vec!["dotnet".to_string()];

    for candidate in &candidates {
        let output = std::process::Command::new(candidate)
            .arg("--list-runtimes")
            .output();
        if let Ok(out) = output {
            let text = String::from_utf8_lossy(&out.stdout).into_owned();
            let version = parse_dotnet_runtimes(&text);
            if version.is_some() {
                let ok = true;
                return (version, ok);
            }
        }
    }
    (None, false)
}
```

```rust
use std::path::{Path, PathBuf};

/// Inject BepInEx doorstop args into LaunchOptions for App ID 2138550.
/// Uses string-level manipulation for robustness with Valve's VDF quirks.
pub fn inject_launch_options(content: &str, args: &str) -> Result<String, String> {
    let marker = "\"2138550\"";
    let lo_key = "\"LaunchOptions\"";

    let app_start = content.find(marker)
        .ok_or("App ID 2138550 not found in localconfig.vdf")?;

    let brace_start = content[app_start..].find('{')
        .map(|i| app_start + i)
        .ok_or("Malformed VDF: no opening brace for 2138550")?;

    let mut depth = 0usize;
    let mut brace_end = None;
    for (i, c) in content[brace_start..].char_indices() {
        match c {
            '{' => depth += 1,
            '}' => {
                depth -= 1;
                if depth == 0 { brace_end = Some(brace_start + i); break; }
            }
            _ => {}
        }
    }
    let brace_end = brace_end.ok_or("Malformed VDF: unmatched brace for 2138550")?;
    let block = &content[brace_start..=brace_end];
    let new_lo_line = format!("\t\t\t\t\t\"LaunchOptions\"\t\t\"{}\"", args);

    let new_block = if let Some(lo_pos) = block.find(lo_key) {
        let line_end = block[lo_pos..].find('\n').map(|i| lo_pos + i).unwrap_or(block.len() - 1);
        format!("{}{}{}", &block[..lo_pos], new_lo_line, &block[line_end..])
    } else {
        let insert_pos = block.rfind('}').ok_or("No closing brace in 2138550 block")?;
        format!("{}{}\n{}", &block[..insert_pos], new_lo_line, &block[insert_pos..])
    };

    Ok(format!("{}{}{}", &content[..brace_start], new_block, &content[brace_end + 1..]))
}

/// Find localconfig.vdf under the first numeric userdata subdirectory.
/// Returns None if no valid path is found or the file doesn't exist.
pub fn find_localconfig_path(steam_path: &Path) -> Option<PathBuf> {
    let userdata = steam_path.join("userdata");
    let lc = std::fs::read_dir(userdata).ok()?
        .filter_map(|e| e.ok())
        .find(|e| e.path().is_dir())
        .map(|e| e.path().join("config/localconfig.vdf"))?;
    if lc.exists() { Some(lc) } else { None }
}

#[tauri::command]
pub fn patch_launch_options(app: tauri::AppHandle, steam_path: String, _game_path: String) -> Result<(), String> {
    use tauri::Emitter;

    #[cfg(target_os = "macos")]
    let args = "DYLD_INSERT_LIBRARIES=./doorstop_libs/libdoorstop.dylib";
    #[cfg(target_os = "windows")]
    let args = "--doorstop-enable true --doorstop-target BepInEx/core/BepInEx.Preloader.dll";
    #[cfg(not(any(target_os = "macos", target_os = "windows")))]
    let args = "";

    app.emit("installer://log", "Locating localconfig.vdf…").ok();
    let lc_path = find_localconfig_path(Path::new(&steam_path))
        .ok_or("Could not find localconfig.vdf under Steam/userdata")?;

    let content = std::fs::read_to_string(&lc_path).map_err(|e| e.to_string())?;

    let backup = lc_path.with_extension("vdf.bak");
    std::fs::copy(&lc_path, &backup).map_err(|e| e.to_string())?;
    app.emit("installer://log", "Backed up localconfig.vdf").ok();

    let modified = inject_launch_options(&content, args)?;

    let tmp = lc_path.with_extension("vdf.tmp");
    std::fs::write(&tmp, modified).map_err(|e| e.to_string())?;
    std::fs::rename(&tmp, &lc_path).map_err(|e| e.to_string())?;
    app.emit("installer://log", "localconfig.vdf updated.").ok();

    Ok(())
}
```

```rust
use std::path::Path;
use std::io::Read;

/// Extract a zip archive to dest_dir. Returns list of extracted file paths.
pub fn extract_zip(zip_bytes: &[u8], dest_dir: &Path) -> Result<Vec<String>, String> {
    let cursor = std::io::Cursor::new(zip_bytes);
    let mut archive = zip::ZipArchive::new(cursor).map_err(|e| e.to_string())?;
    let mut extracted = Vec::new();
    for i in 0..archive.len() {
        let mut file = archive.by_index(i).map_err(|e| e.to_string())?;
        let outpath = dest_dir.join(file.name());
        if file.is_dir() {
            std::fs::create_dir_all(&outpath).map_err(|e| e.to_string())?;
        } else {
            if let Some(parent) = outpath.parent() {
                std::fs::create_dir_all(parent).map_err(|e| e.to_string())?;
            }
            let mut contents = Vec::new();
            file.read_to_end(&mut contents).map_err(|e| e.to_string())?;
            std::fs::write(&outpath, contents).map_err(|e| e.to_string())?;
            extracted.push(outpath.to_string_lossy().into_owned());
        }
    }
    Ok(extracted)
}

#[tauri::command]
pub fn install_bepinex(app: tauri::AppHandle, game_path: String) -> Result<(), String> {
    use tauri::Emitter;
    app.emit("installer://log", "Reading bundled BepInEx.zip…").ok();

    let resource_path = app.path().resource_dir()
        .map_err(|e| e.to_string())?
        .join("BepInEx.zip");

    let zip_bytes = std::fs::read(&resource_path)
        .map_err(|e| format!("Cannot read bundled BepInEx.zip: {}", e))?;

    app.emit("installer://log", "Extracting BepInEx…").ok();
    let files = extract_zip(&zip_bytes, Path::new(&game_path))?;
    app.emit("installer://log", format!("Extracted {} files.", files.len())).ok();
    Ok(())
}
```

```svelte
  import { listen } from "@tauri-apps/api/event";
  import { open } from "@tauri-apps/plugin-dialog";
  import { onMount } from "svelte";
  import type { EnvironmentInfo, CardStatus } from "./lib/types";

  let env: EnvironmentInfo | null = null;
  let loading = true;
  let installing = false;
  let logs: string[] = [];
  let customGamePath = "";
  let logEl: HTMLElement;

  function effectiveGamePath(): string {
    return customGamePath || env?.game_path || "";
  }

  function cardStatus(value: string | null | boolean, isOk?: boolean): CardStatus {
    if (loading) return "loading";
    if (typeof isOk === "boolean") return isOk ? "ok" : "warn";
    return value ? "ok" : "error";
  }

  function addLog(msg: string) {
    logs = [...logs, `[${new Date().toLocaleTimeString()}] ${msg}`];
    setTimeout(() => logEl?.scrollTo(0, logEl.scrollHeight), 10);
  }

  onMount(async () => {
    await listen("installer://log", (event) => {
      addLog(event.payload as string);
    });

    try {
      env = await invoke<EnvironmentInfo>("detect_environment");
      if (env.game_path) customGamePath = "";
    } catch (e) {
      addLog("Detection error: " + e);
    } finally {
      loading = false;
    }
  });

  async function pickGamePath() {
    const selected = await open({ directory: true, multiple: false });
    if (typeof selected === "string") {
      customGamePath = selected;
    } else if (Array.isArray(selected) && selected.length > 0) {
      customGamePath = selected[0];
    }
  }

  async function install() {
    installing = true;
    addLog("Starting installation...");
    try {
      await invoke("install_bepinex", { gamePath: effectiveGamePath() });
      addLog("BepInEx extracted.");
      await invoke("patch_launch_options", {
        steamPath: env?.steam_path ?? "",
        gamePath: effectiveGamePath(),
      });
      addLog("Steam launch options patched.");
      addLog("✓ Installation complete! Launch the game from Steam.");
    } catch (e) {
      addLog("✗ Error: " + e);
    } finally {
      installing = false;
    }
  }
```
